    """세션 dict → 노트 파일 경로 (프로젝트 하위폴더/날짜 제목 (id8).md)"""
    date = to_date(sess["first_ts"])

    return Path(out_root) / sanitize(sess["project"] or "unknown", 40) / fname


def convert_one(jsonl_path: Path, out_root: Path,
                max_chars: int = 40000, min_prompts: int = 1) -> Path | None:
    """단일 세션 변환. 훅(archive_session.py)에서 사용. 조건 미달이면 None."""

    target = note_target(sess, Path(out_root))
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_text(render_note(sess, max_chars), encoding="utf-8")
    return target

```python
#!/usr/bin/env python3
"""볼트 비밀정보 스캔. HIGH 발견 시 exit 1 (CI 게이트 겸용).

사용:  python scripts/secret_scan.py vault [다른경로...]
"""
from __future__ import annotations

import re
import sys
from pathlib import Path

if sys.stdout and hasattr(sys.stdout, "reconfigure"):
    sys.stdout.reconfigure(encoding="utf-8", errors="replace")
    sys.stderr.reconfigure(encoding="utf-8", errors="replace")

PATTERNS: list[tuple[str, str, re.Pattern]] = [
    ("HIGH", "anthropic-key", re.compile(r"sk-ant-[A-Za-z0-9_-]{10,}")),
    ("HIGH", "openai-key", re.compile(r"sk-[A-Za-z0-9]{20}T3BlbkFJ[A-Za-z0-9]{20}")),
    ("HIGH", "aws-access-key", re.compile(r"AKIA[0-9A-Z]{16}")),
    ("HIGH", "github-token", re.compile(r"gh[pousr]_[A-Za-z0-9]{36,}")),
    ("HIGH", "private-key", re.compile(r"-----BEGIN [A-Z ]*PRIVATE KEY-----")),
    ("MEDIUM", "bearer-token", re.compile(r"Bearer\s+[A-Za-z0-9._\-]{20,}")),
    ("MEDIUM", "generic-secret",
     re.compile(r"(?i)(?:api[_-]?key|secret|password|token)\s*[:=]\s*[\"'][^\"']{12,}[\"']")),
]

SCAN_EXT = {".md", ".txt", ".json", ".yaml", ".yml"}
SKIP_DIRS = {".git", ".obsidian", "node_modules", "__pycache__"}


def _mask(s: str) -> str:
    return s[:8] + "…"


def scan_paths(paths: list[Path]) -> list[tuple[str, str, Path, int, str]]:
    findings = []
    for root in paths:
        files = [root] if root.is_file() else sorted(
            p for p in root.rglob("*")
            if p.is_file() and p.suffix.lower() in SCAN_EXT
            and not (SKIP_DIRS & set(p.parts))
        )
        for f in files:
            try:
                text = f.read_text(encoding="utf-8", errors="replace")
            except OSError:
                continue
            for i, line in enumerate(text.splitlines(), 1):
                for sev, rule, rx in PATTERNS:
                    m = rx.search(line)
                    if m:
                        findings.append((sev, rule, f, i, _mask(m.group(0))))
    return findings


def main() -> int:
    targets = [Path(a) for a in sys.argv[1:]] or [Path("vault")]
    findings = scan_paths(targets)
    high = [f for f in findings if f[0] == "HIGH"]
    for sev, rule, f, line, masked in findings:
        print(f"[{sev}] {rule}  {f}:{line}  {masked}")
    print(f"\n스캔 완료: HIGH {len(high)}건, MEDIUM {len(findings) - len(high)}건")
    if high:
        print("HIGH 발견 — 해당 키를 폐기(rotate)하고 노트에서 제거하세요. 배포 차단됨.")
        return 1
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

import json
import subprocess
import sys
import tempfile
import unittest
from pathlib import Path

SCRIPT = Path(__file__).parents[1] / "archive_session.py"
FIXTURE = Path(__file__).parent / "fixtures" / "sample.jsonl"


class TestHook(unittest.TestCase):
    def _run(self, payload: dict, out_dir: str):
        return subprocess.run(
            [sys.executable, str(SCRIPT), "--out", out_dir],
            input=json.dumps(payload), capture_output=True, text=True, timeout=30,

            self.assertEqual(r.returncode, 0)

            r = self._run({}, td)

        with tempfile.TemporaryDirectory() as td:
            r = subprocess.run(
                [sys.executable, str(SCRIPT), "--out", td],
                input="not json", capture_output=True, text=True, timeout=30,
            )

- [ ] **Step 3: 구현** — `scripts/archive_session.py`:

```python
#!/usr/bin/env python3
"""SessionEnd 훅: 끝난 세션을 볼트 90-Archive에 md로 아카이브.

stdin으로 훅 페이로드 JSON(transcript_path 포함)을 받는다.
어떤 경우에도 exit 0 — 세션 종료를 절대 블로킹하지 않는다(best-effort).
"""
from __future__ import annotations

import argparse
import json
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent))
from cc2obsidian import convert_one  # noqa: E402

DEFAULT_OUT = Path(__file__).resolve().parents[1] / "vault" / "90-Archive" / "claude-sessions"


def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("--out", default=str(DEFAULT_OUT))
    args = ap.parse_args()
    out_root = Path(args.out)
    try:
        payload = json.load(sys.stdin)
        tp = payload.get("transcript_path")
        if tp and Path(tp).is_file():
            convert_one(Path(tp), out_root, min_prompts=2)
    except Exception as e:  # noqa: BLE001 — 훅은 무조건 성공 종료
        try:
            out_root.mkdir(parents=True, exist_ok=True)
            with (out_root / "_hook-errors.log").open("a", encoding="utf-8") as f:
                f.write(f"{e!r}\n")
        except OSError:
            pass
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
